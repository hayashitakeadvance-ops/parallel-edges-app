    const questionEl = document.getElementById('question');
    const shelfEl = document.getElementById('shelf');
    const messageEl = document.getElementById('message');
    const pointsEl = document.getElementById('points');
    const hanamaruEl = document.getElementById('hanamaru');

    const praiseMessages = ['せいかい！', 'かしこい！', 'てんさい！', 'すごい！', '１００点！'];

    const fruitPool = [
      { key: 'banana', name: 'バナナ' },
      { key: 'strawberry', name: 'いちご' },
      { key: 'melon', name: 'メロン' },
      { key: 'grape', name: 'ぶどう' },
      { key: 'watermelon', name: 'すいか' },
      { key: 'apple', name: 'りんご' },
      { key: 'orange', name: 'みかん' }
    ];

    const state = {
      fruits: [],
      answerIndex: 0,
      selectedIndex: null,
      fromSide: 'left',
      points: 0,
      locked: false
    };

    function shuffle(arr) {
      const a = [...arr];
      for (let i = a.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [a[i], a[j]] = [a[j], a[i]];
      }
      return a;
    }

    function makeProblem() {
      const counts = [3, 4, 5, 6, 7];
      const count = counts[Math.floor(Math.random() * counts.length)];
      const visiblePool = [
        { key: 'banana', name: 'バナナ' },
        { key: 'strawberry', name: 'いちご' },
        { key: 'melon', name: 'メロン' },
        { key: 'grape', name: 'ぶどう' },
        { key: 'watermelon', name: 'すいか' },
        { key: 'apple', name: 'りんご' },
        { key: 'orange', name: 'みかん' }
      ];

      state.fruits = shuffle(visiblePool).slice(0, count);

      while (state.fruits.length < 3) {
        state.fruits = shuffle(visiblePool).slice(0, 3);
      }

      state.answerIndex = Math.floor(Math.random() * state.fruits.length);
      state.fromSide = Math.random() < 0.5 ? 'left' : 'right';
      state.selectedIndex = null;
      state.locked = false;
    }

    function getQuestionNumber() {
      return state.fromSide === 'left'
        ? state.answerIndex + 1
        : state.fruits.length - state.answerIndex;
    }

    function updateQuestion() {
      const fruitName = state.fruits[state.answerIndex].name;
      const sideText = state.fromSide === 'left' ? 'ひだり' : 'みぎ';
      const num = getQuestionNumber();
      questionEl.innerHTML =
        `<span class="questionBox"><span class="hiddenWord" aria-label="${fruitName}">${fruitName}</span><span class="smallText">は、</span>${sideText}<span class="smallText">から</span> ${num}ばん<span class="redChar">め</span></span><span class="touchText"> を さわろう</span>`;
    }

    function updateMessage() {
      if (state.selectedIndex === null) {
        messageEl.className = 'message';
        messageEl.textContent = '';
        return;
      }

      if (state.selectedIndex === state.answerIndex) {
        state.locked = true;
        state.points += 1;
        pointsEl.textContent = `${state.points}ポイント`;
        messageEl.className = 'message ok';
        messageEl.textContent = praiseMessages[Math.floor(Math.random() * praiseMessages.length)];

        if (state.points > 0 && state.points % 5 === 0) {
          hanamaruEl.style.display = 'flex';
          setTimeout(() => {
            hanamaruEl.style.display = 'none';
          }, 700);
        }

        setTimeout(() => {
          nextProblem();
        }, 900);
      } else {
        messageEl.className = 'message ng';
        messageEl.textContent = 'おしい！';
      }
    }

    function fruitSVG(key) {
      if (key === 'banana') {

          </svg>`;
      }

      if (key === 'strawberry') {
        return `
          <svg class="fruitSvg" viewBox="0 0 120 110" xmlns="http://www.w3.org/2000/svg" aria-hidden="true">
            <path d="M58 24 C45 24, 34 35, 34 51 C34 73, 49 92, 60 92 C71 92, 86 73, 86 51 C86 35, 75 24, 62 24 Z" fill="#f53a30"/>
            <path d="M48 23 L60 14 L72 23" fill="#2dbb5d"/>
            <circle cx="52" cy="44" r="1.8" fill="#7a2a17"/>
            <circle cx="63" cy="41" r="1.8" fill="#7a2a17"/>
            <circle cx="71" cy="50" r="1.8" fill="#7a2a17"/>
            <circle cx="56" cy="56" r="1.8" fill="#7a2a17"/>
            <circle cx="67" cy="61" r="1.8" fill="#7a2a17"/>
          </svg>`;
      }

      if (key === 'melon') {
        return `
          <svg class="fruitSvg" viewBox="0 0 120 110" xmlns="http://www.w3.org/2000/svg" aria-hidden="true">
            <circle cx="60" cy="55" r="34" fill="#dfe8da"/>
            <path d="M60 21 C58 18, 58 14, 62 12 C65 14, 65 18, 63 21" fill="#caa84d"/>
            <path d="M35 40 C48 28, 73 27, 84 39 C90 46, 93 56, 88 68 C80 84, 60 89, 45 82 C30 75, 25 56, 35 40 Z" fill="none" stroke="#b7c5b0" stroke-width="2"/>
            <path d="M33 53 C45 44, 77 44, 89 55" fill="none" stroke="#b7c5b0" stroke-width="2"/>
            <path d="M41 26 C49 38, 50 74, 44 84" fill="none" stroke="#b7c5b0" stroke-width="2"/>
            <path d="M61 22 C64 37, 64 73, 60 88" fill="none" stroke="#b7c5b0" stroke-width="2"/>
            <path d="M79 29 C72 43, 72 70, 79 82" fill="none" stroke="#b7c5b0" stroke-width="2"/>
          </svg>`;
      }

      if (key === 'grape') {
        return `
          <svg class="fruitSvg" viewBox="0 0 120 110" xmlns="http://www.w3.org/2000/svg" aria-hidden="true">
            <rect x="68" y="13" width="8" height="10" rx="2" fill="#89612b"/>

            <circle cx="64" cy="44" r="11" fill="#597cf0"/>
            <circle cx="78" cy="47" r="11" fill="#4b6cda"/>
            <circle cx="56" cy="58" r="11" fill="#6281ee"/>
            <circle cx="70" cy="59" r="11" fill="#5376e2"/>
            <circle cx="62" cy="73" r="11" fill="#6e8df3"/>
          </svg>`;
      }

      if (key === 'watermelon') {
        return `
          <svg class="fruitSvg" viewBox="0 0 120 110" xmlns="http://www.w3.org/2000/svg" aria-hidden="true">
            <path d="M25 68 A36 36 0 0 1 95 68 Z" fill="#16a65a"/>
            <path d="M31 66 A30 30 0 0 1 89 66 Z" fill="#ffffff"/>
            <path d="M35 63 A26 26 0 0 1 85 63 Z" fill="#f53b30"/>
            <ellipse cx="48" cy="57" rx="2" ry="4" fill="#1b1b1b"/>
            <ellipse cx="60" cy="53" rx="2" ry="4" fill="#1b1b1b"/>
            <ellipse cx="73" cy="58" rx="2" ry="4" fill="#1b1b1b"/>
          </svg>`;
      }

      if (key === 'orange') {
        return `
          <svg class="fruitSvg" viewBox="0 0 120 110" xmlns="http://www.w3.org/2000/svg" aria-hidden="true">
            <circle cx="60" cy="56" r="28" fill="#ff9c2e"/>
            <circle cx="52" cy="50" r="10" fill="#ffb84f" opacity="0.55"/>
            <path d="M63 28 C63 20, 69 15, 76 15" stroke="#9a5a22" stroke-width="5" fill="none" stroke-linecap="round"/>

          </svg>`;
      }

      if (key === 'apple') {
        return `
          <svg class="fruitSvg" viewBox="0 0 120 110" xmlns="http://www.w3.org/2000/svg" aria-hidden="true">
            <circle cx="60" cy="56" r="30" fill="#ff4438"/>
            <path d="M63 24 C63 16, 69 11, 76 11" stroke="#9a5a22" stroke-width="5" fill="none" stroke-linecap="round"/>

          </svg>`;
      }

      return `<svg class="fruitSvg" viewBox="0 0 120 110" xmlns="http://www.w3.org/2000/svg" aria-hidden="true"></svg>`;
    }

    function renderShelf() {
      shelfEl.innerHTML = '';
      state.fruits.forEach((fruit, index) => {
        const btn = document.createElement('button');
        btn.className = 'fruitBtn' + (state.selectedIndex === index ? ' selected' : '');
        btn.innerHTML = fruitSVG(fruit.key);

        btn.addEventListener('click', () => {
          if (state.locked) return;
          state.selectedIndex = index;
          renderShelf();
          updateMessage();
        });

        shelfEl.appendChild(btn);
      });
    }

    function nextProblem() {
      makeProblem();
      updateQuestion();
      renderShelf();
      messageEl.className = 'message';
      messageEl.textContent = '';
      pointsEl.textContent = `${state.points}ポイント`;
    }

    nextProblem();
  </script>
